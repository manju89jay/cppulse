cmake_minimum_required(VERSION 3.16)
project(cppulse-first-trial LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(trial-checks checks.cpp)
target_compile_options(trial-checks PRIVATE -Wall -Wextra -Werror)
enable_testing()
foreach(case negative zero positive)
    add_test(NAME clamp-${case} COMMAND trial-checks ${case})
endforeach()
