cmake_minimum_required(VERSION 3.16)
project(cppulse-real-consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CPPULSE_CORPUS_ROOT)
  message(FATAL_ERROR "Pass CPPULSE_CORPUS_ROOT with pinned fmt and utfcpp checkouts")
endif()
set(FMT_TEST OFF CACHE BOOL "" FORCE)
set(FMT_DOC OFF CACHE BOOL "" FORCE)
add_subdirectory("${CPPULSE_CORPUS_ROOT}/fmt" fmt-build EXCLUDE_FROM_ALL)
add_subdirectory("${CPPULSE_CORPUS_ROOT}/utfcpp" utfcpp-build EXCLUDE_FROM_ALL)
add_executable(cppulse-consumer main.cpp)
target_link_libraries(cppulse-consumer PRIVATE fmt::fmt utf8cpp)
enable_testing()
add_test(NAME utf8-format-consumer COMMAND cppulse-consumer)
